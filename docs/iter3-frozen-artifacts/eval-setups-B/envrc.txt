# --- Cato VPN TLS interception ---
# Cato Networks injects its own root CA; build certifi+Cato bundle once and
# point urllib3 / requests / transformers / curl at it.
_RC_CA_BUNDLE="$HOME/.cache/reasoning-core/ca-bundle.pem"
if [[ ! -s "$_RC_CA_BUNDLE" ]] && command -v security >/dev/null 2>&1; then
  mkdir -p "$(dirname "$_RC_CA_BUNDLE")"
  _RC_CERTIFI="$(python3 -c 'import certifi; print(certifi.where())' 2>/dev/null)"
  if [[ -f "$_RC_CERTIFI" ]]; then
    cat "$_RC_CERTIFI" > "$_RC_CA_BUNDLE"
    security find-certificate -a -p -c "Cato Networks" /Library/Keychains/System.keychain >> "$_RC_CA_BUNDLE" 2>/dev/null || true
  fi
fi
if [[ -s "$_RC_CA_BUNDLE" ]]; then
  export SSL_CERT_FILE="$_RC_CA_BUNDLE"
  export REQUESTS_CA_BUNDLE="$_RC_CA_BUNDLE"
  export CURL_CA_BUNDLE="$_RC_CA_BUNDLE"
fi
unset _RC_CA_BUNDLE _RC_CERTIFI

# Lazy-load the API key only when scw is available (CI Linux skips it).
# Symmetric unset (engineer v4 review): if scw is absent OR returns empty,
# explicitly clear RC_GEN_API_KEY so a stale rotated/revoked key from a
# prior load cannot persist. Mirrors the ANTHROPIC_AUTH_TOKEN pattern in
# reasoning-core/.envrc.
if command -v scw >/dev/null 2>&1; then
  RC_GEN_API_KEY_VAL="$(scw config get secret-key --profile circit 2>/dev/null)"
  if [[ -n "$RC_GEN_API_KEY_VAL" ]]; then
    export RC_GEN_API_KEY="$RC_GEN_API_KEY_VAL"
  else
    unset RC_GEN_API_KEY
  fi
  unset RC_GEN_API_KEY_VAL
else
  unset RC_GEN_API_KEY

# --- Iter-3 lever self-check (newsletter v4 review) ---
# Catch env-var theatre: lever set to 1 but corresponding hook never
# registered in this folder's settings.local.json. WARN loudly at direnv
# load so an operator notices BEFORE running the eval.
_RC_SETTINGS_FILE="$(dirname "${BASH_SOURCE[0]}")/settings.local.json"
if [[ "$RC_BEST_EFFORT_SPEC" == "1" ]] && [[ -f "$_RC_SETTINGS_FILE" ]]; then
  # Tighter pattern (engineer round-3 review): match the actual hook command
  # entry shape, not just the substring — avoids false-positives on filenames
  # like `session_start_best_effort_disabled.py` or a JSON `_note` key that
  # happens to contain the substring.
  if ! grep -qE '"command"[^"]*"[^"]*session_start_best_effort\.py"' "$_RC_SETTINGS_FILE" 2>/dev/null; then
    echo "[direnv] WARN: RC_BEST_EFFORT_SPEC=1 but session_start_best_effort.py not registered as a hook command in $_RC_SETTINGS_FILE — lever is INERT"
  fi
fi
unset _RC_SETTINGS_FILE
